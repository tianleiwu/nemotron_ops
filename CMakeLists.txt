cmake_minimum_required(VERSION 3.22)
set(CMAKE_CUDA_ARCHITECTURES OFF)
project(nemotron_ops LANGUAGES CXX CUDA)
set(ORT_INCLUDE_DIR "" CACHE PATH "Directory containing onnxruntime_cxx_api.h")
if(NOT EXISTS "${ORT_INCLUDE_DIR}/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "set -DORT_INCLUDE_DIR=<ort>/include/onnxruntime/core/session")
endif()
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CUDA_FLAGS_RELEASE "-O3")
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(bolt_ops SHARED src/bolt_ops.cu)
option(BOLT_MAMBA_ROUND_CONV_BEFORE_SILU "Round the Mamba convolution result to BF16 before SiLU" OFF)
option(BOLT_MAMBA_ROUND_READOUT_BEFORE_NORM "Round the Mamba SSM readout to BF16 before gated normalization" OFF)
target_compile_definitions(bolt_ops PRIVATE
  BOLT_MAMBA_ROUND_CONV_BEFORE_SILU=$<BOOL:${BOLT_MAMBA_ROUND_CONV_BEFORE_SILU}>
  BOLT_MAMBA_ROUND_READOUT_BEFORE_NORM=$<BOOL:${BOLT_MAMBA_ROUND_READOUT_BEFORE_NORM}>)
target_include_directories(bolt_ops PRIVATE "${ORT_INCLUDE_DIR}")
target_compile_options(bolt_ops PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-arch=sm_90;-lineinfo>")
set_target_properties(bolt_ops PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

option(BOLT_BUILD_MAMBA_ROUNDING_TESTS "Build focused CUDA Mamba rounding regressions" OFF)
if(BOLT_BUILD_MAMBA_ROUNDING_TESTS)
  enable_testing()
  foreach(conv_round IN ITEMS 0 1)
    foreach(readout_round IN ITEMS 0 1)
      set(test_name mamba_rounding_${conv_round}_${readout_round})
      add_executable(${test_name} tests/mamba_rounding_test.cu)
      target_include_directories(${test_name} PRIVATE "${ORT_INCLUDE_DIR}")
      target_compile_options(${test_name} PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-arch=sm_90>")
      target_compile_definitions(${test_name} PRIVATE
        BOLT_MAMBA_ROUND_CONV_BEFORE_SILU=${conv_round}
        BOLT_MAMBA_ROUND_READOUT_BEFORE_NORM=${readout_round})
      add_test(NAME ${test_name} COMMAND ${test_name})
      set_tests_properties(${test_name} PROPERTIES TIMEOUT 120)
    endforeach()
  endforeach()
endif()
